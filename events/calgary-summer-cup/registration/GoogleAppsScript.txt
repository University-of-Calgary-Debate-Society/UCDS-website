function doPost(e) {
  // CORS configuration response headers
  var headersResponse = {
    "Access-Control-Allow-Origin": "*",
    "Access-Control-Allow-Methods": "POST, GET, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type"
  };

  try {
    var p = e.parameter;

    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    // Set up headers if the sheet is completely empty
    var headers = ["Timestamp", "Full Name", "Email", "Grade (2025-2026)", "Partner Name", "Partner Email", "Partner Grade (2025-2026)", "Category", "Suggestions"];
    if (sheet.getLastRow() === 0) {
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
      sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold");
      sheet.setFrozenRows(1);
    }
    
    // Parse parameters
    var rowData = [
      new Date(),
      p.fullName || "",
      p.email || "",
      p.grade || "",
      p.partnerName || "",
      p.partnerEmail || "",
      p.partnerGrade || "",
      p.category || "",
      p.suggestions || ""
    ];
    
    sheet.appendRow(rowData);
    
    // Send confirmation email to the registrant
    if (p.email) {
      try {
        var subject = "Registration Confirmed: Calgary Summer Cup 2026";
        
        // Plain text fallback
        var body = "Hi " + (p.fullName || "there") + ",\n\n" +
                   "Thank you for registering for the Calgary Summer Cup 2026!\n\n" +
                   "We have received your registration. Here are the details we recorded:\n" +
                   " - Your Name: " + (p.fullName || "") + "\n" +
                   " - Your Grade: " + (p.grade || "") + "\n" +
                   " - Partner's Name: " + (p.partnerName || "") + "\n" +
                   " - Partner's Email: " + (p.partnerEmail || "") + "\n" +
                   " - Partner's Grade: " + (p.partnerGrade || "") + "\n" +
                   " - Selected Category: " + (p.category || "") + "\n\n" +
                   "If you need to update any information or have questions, please reply to this email.\n\n" +
                   "Best regards,\n" +
                   "University of Calgary Debate Society (UCDS)";
        
        // Beautiful HTML email matching UCDS branding that blends seamlessly with the inbox
        var htmlBody = 
          '<div style="font-family: -apple-system, BlinkMacSystemFont, \'Segoe UI\', Roboto, Helvetica, Arial, sans-serif; max-width: 560px; margin: 0 auto; padding: 20px 0; color: #1e293b; line-height: 1.5; text-align: left;">' +
            '<div style="border-bottom: 2px solid #112854; padding-bottom: 12px; margin-bottom: 24px;">' +
              '<h3 style="color: #112854; margin: 0; font-size: 1.05rem; letter-spacing: 0.05em; font-weight: bold; text-transform: uppercase;">University of Calgary Debate Society</h3>' +
            '</div>' +
            '<div style="text-align: center; margin-bottom: 20px;">' +
              '<img src="https://lh3.googleusercontent.com/d/1UZ8Necs4jC0xV-WzM8L6n8pTRer522El" alt="Calgary Summer Cup Logo" style="max-height: 100px; width: auto; display: inline-block; filter: drop-shadow(0 4px 10px rgba(0,0,0,0.15));" />' +
            '</div>' +
            '<h2 style="color: #16a34a; font-size: 1.6rem; margin-top: 0; margin-bottom: 8px; font-weight: 700;">You\'re Registered!</h2>' +
            '<p style="font-size: 1rem; color: #334155; margin-bottom: 24px;">' +
               'Thank you for registering for the <strong>Calgary Summer Cup 2026</strong>. We have successfully recorded your details.' +
            '</p>' +
            '<div style="background-color: #f8fafc; border: 1px solid #e2e8f0; border-radius: 8px; padding: 20px; margin-bottom: 24px;">' +
              '<h4 style="color: #112854; margin-top: 0; margin-bottom: 12px; font-size: 0.95rem; text-transform: uppercase; letter-spacing: 0.02em;">Registration Receipt</h4>' +
              '<table style="width: 100%; border-collapse: collapse; font-size: 0.95rem;">' +
                '<tr>' +
                  '<td style="padding: 6px 0; font-weight: 600; width: 40%; color: #64748b;">Your Name:</td>' +
                  '<td style="padding: 6px 0; color: #0f172a;">' + (p.fullName || "") + '</td>' +
                '</tr>' +
                '<tr>' +
                  '<td style="padding: 6px 0; font-weight: 600; color: #64748b;">Your Grade:</td>' +
                  '<td style="padding: 6px 0; color: #0f172a;">' + (p.grade || "") + '</td>' +
                '</tr>' +
                '<tr>' +
                  '<td style="padding: 6px 0; font-weight: 600; color: #64748b;">Partner\'s Name:</td>' +
                  '<td style="padding: 6px 0; color: #0f172a;">' + (p.partnerName || "") + '</td>' +
                '</tr>' +
                '<tr>' +
                  '<td style="padding: 6px 0; font-weight: 600; color: #64748b;">Partner\'s Email:</td>' +
                  '<td style="padding: 6px 0; color: #0f172a;">' + (p.partnerEmail || "") + '</td>' +
                '</tr>' +
                '<tr>' +
                  '<td style="padding: 6px 0; font-weight: 600; color: #64748b;">Partner\'s Grade:</td>' +
                  '<td style="padding: 6px 0; color: #0f172a;">' + (p.partnerGrade || "") + '</td>' +
                '</tr>' +
                '<tr>' +
                  '<td style="padding: 6px 0; font-weight: 600; color: #64748b;">Category:</td>' +
                  '<td style="padding: 6px 0; color: #16a34a; font-weight: bold;">' + (p.category || "") + '</td>' +
                '</tr>' +
              '</table>' +
            '</div>' +
            '<p style="font-size: 0.95rem; color: #334155; margin-bottom: 24px;">' +
              'The tournament will run online over <strong>Discord</strong> on <strong>July 25th & 26th</strong>. Keep an eye on your inbox for official server links, judge briefings, and checking times as the tournament approaches.' +
            '</p>' +
            '<div style="border-top: 1px solid #e2e8f0; padding-top: 16px; font-size: 0.85rem; color: #64748b; line-height: 1.6;">' +
              'If you need to make changes or have questions, reply directly to this email or contact us at <a href="mailto:ucdebate@ucalgary.ca" style="color: #2563eb; text-decoration: none;">ucdebate@ucalgary.ca</a>.<br>' +
              '&copy; 2026 University of Calgary Debate Society. All rights reserved.' +
            '</div>' +
          '</div>';

        MailApp.sendEmail({
          to: p.email,
          subject: subject,
          body: body,
          htmlBody: htmlBody
        });
      } catch (mailError) {
        console.error("Failed to send confirmation email: " + mailError.toString());
      }
    }
    
    return ContentService.createTextOutput(JSON.stringify({ result: "success" }))
                         .setMimeType(ContentService.MimeType.JSON);
                         
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ result: "error", error: error.toString() }))
                         .setMimeType(ContentService.MimeType.JSON);
  }
}
