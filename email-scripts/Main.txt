// =========================================================================
// ENTRY POINTS (doPost and doGet HTTP triggers)
// =========================================================================

function doPost(e) {
  var payload = null;
  var p = {};
  
  // Try parsing JSON POST payload (useful for Python worker calls and ajax POSTs)
  if (e && e.postData && e.postData.contents) {
    try {
      payload = JSON.parse(e.postData.contents);
      p = payload;
    } catch(err) {
      // Fallback if content is not JSON
    }
  }
  
  // Merge URL query parameter values if any
  if (e && e.parameter) {
    for (var key in e.parameter) {
      p[key] = e.parameter[key];
    }
  }

  // 1. Check if it's a subscribe action (public, no authentication needed)
  if (p.action === "subscribe" || (!p.action && p.email)) {
    try {
      return handleSubscribe(p);
    } catch (error) {
      return jsonResponse({ result: "error", error: error.toString() });
    }
  }

  // 2. Admin actions (authentication required)
  if (!p.password || p.password !== ADMIN_PASSWORD) {
    return jsonResponse({ result: "error", error: "Unauthorized" });
  }

  try {
    var spreadsheet = getMailingListSpreadsheet();
    
    // Route POST actions to their handlers
    if (p.action === "saveDraft") {
      return handleSaveDraft(spreadsheet, p);
    }
    if (p.action === "deleteDraft") {
      return handleDeleteDraft(spreadsheet, p);
    }
    if (p.action === "scheduleSend") {
      return handleScheduleSend(spreadsheet, p);
    }
    if (p.action === "cancelSchedule") {
      return handleCancelSchedule(spreadsheet, p);
    }
    if (p.action === "markSent") {
      return handleMarkSent(spreadsheet, p);
    }
    if (p.action === "getDrafts") {
      return handleGetDrafts(spreadsheet);
    }
    if (p.action === "getMailingLists") {
      return handleGetMailingLists(spreadsheet);
    }
    if (p.action === "getSchedules") {
      return handleGetSchedules(spreadsheet);
    }
    if (p.action === "getPendingSends") {
      return handleGetPendingSends(spreadsheet);
    }
    if (p.action === "sendTestEmail") {
      return handleSendTestEmail(spreadsheet, p);
    }
    if (p.action === "sendImmediately") {
      return handleSendImmediately(spreadsheet, p);
    }
    if (p.action === "createDummy") {
      return handleCreateDummy(spreadsheet, p);
    }
    
    return jsonResponse({ result: "error", error: "Invalid action" });
  } catch (error) {
    return jsonResponse({ result: "error", error: error.toString() });
  }
}

function doGet(e) {
  var p = e.parameter;
  
  // Public Unsubscribe flow
  if (p.action === "unsubscribe" && p.email) {
    return handleUnsubscribeRequest(p);
  }
  
  // Public Resubscribe flow
  if (p.action === "resubscribe" && p.email) {
    return handleResubscribeRequest(p);
  }

  // Admin GET endpoints (Require password check)
  if (!p.password || p.password !== ADMIN_PASSWORD) {
    return jsonResponse({ result: "error", error: "Unauthorized" });
  }

  try {
    var spreadsheet = getMailingListSpreadsheet();

    // Route GET actions to their handlers
    if (p.action === "getDrafts") {
      return handleGetDrafts(spreadsheet);
    }
    if (p.action === "getMailingLists") {
      return handleGetMailingLists(spreadsheet);
    }
    if (p.action === "getSchedules") {
      return handleGetSchedules(spreadsheet);
    }
    if (p.action === "getPendingSends") {
      return handleGetPendingSends(spreadsheet);
    }

    return jsonResponse({ result: "error", error: "Invalid action" });
  } catch (error) {
    return jsonResponse({ result: "error", error: error.toString() });
  }
}
