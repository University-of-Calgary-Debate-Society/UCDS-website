/**
 * Standard professional UCDS newsletter template.
 * Generates an outer envelope wrapping the draft body content, complete with floating speech bubbles on the sides.
 */
function compileEmailTemplate(recipientName, messageBody, unsubscribeLink, includeHeader, includeFooter) {
  if (includeHeader === undefined) includeHeader = true;
  if (includeFooter === undefined) includeFooter = true;
  
  var headerHtml = '';
  if (includeHeader === true || includeHeader === "true" || includeHeader === 1) {
    headerHtml = 
      '<!-- Banner Header -->' +
      '<table class="card-table" border="0" cellpadding="0" cellspacing="0" width="600" style="max-width: 600px; background: linear-gradient(135deg, #0b1a3a 0%, #112854 100%); border-radius: 16px; border: 1px solid #0f2249; box-shadow: 0 10px 25px rgba(11, 26, 58, 0.15); overflow: hidden; margin-bottom: 20px;">' +
        '<tr>' +
          '<td align="center" style="padding: 35px 25px; text-align: center;">' +
            '<h1 style="margin: 0 0 10px; color: #ffffff; font-size: 28px; font-weight: 800; letter-spacing: -0.01em; line-height: 1.2;">University of Calgary Debate Society</h1>' +
            '<p style="margin: 0; color: #93c5fd; font-size: 14px; font-weight: 600; text-transform: uppercase; letter-spacing: 0.05em;">Official Newsletter</p>' +
          '</td>' +
        '</tr>' +
      '</table>';
  }
  
  var footerHtml = '';
  if (includeFooter === true || includeFooter === "true" || includeFooter === 1) {
    footerHtml = 
      '<!-- Footer Card -->' +
      '<table class="card-table" border="0" cellpadding="0" cellspacing="0" width="600" style="max-width: 600px; text-align: center;">' +
        '<tr>' +
          '<td style="padding: 10px 20px; font-size: 12px; color: #64748b; line-height: 1.5;">' +
            'You received this email because you are subscribed to the UCDS newsletter list.<br>' +
            '<a href="' + unsubscribeLink + '" style="color: #2563eb; text-decoration: underline; font-weight: 600;">Unsubscribe from this list</a>' +
          '</td>' +
        '</tr>' +
      '</table>';
  } else {
    footerHtml = 
      '<div style="text-align: center; margin-top: 15px; font-size: 11px;">' +
        '<a href="' + unsubscribeLink + '" style="color: #64748b; text-decoration: underline;">Unsubscribe</a>' +
      '</div>';
  }

  if ((includeHeader === false || includeHeader === "false" || includeHeader === 0) &&
      (includeFooter === false || includeFooter === "false" || includeFooter === 0)) {
    return '<!DOCTYPE html>' +
           '<html>' +
           '<head>' +
             '<meta charset="utf-8">' +
             '<meta name="viewport" content="width=device-width, initial-scale=1.0">' +
             '<style>body { font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, Arial, sans-serif; padding: 20px; color: #334155; }</style>' +
           '</head>' +
           '<body>' +
             '<div>' + messageBody + '</div>' +
             '<hr style="border: 0; border-top: 1px solid #e2e8f0; margin: 20px 0;">' +
             '<div style="text-align: center; font-size: 11px;">' +
               '<a href="' + unsubscribeLink + '" style="color: #64748b; text-decoration: underline;">Unsubscribe</a>' +
             '</div>' +
           '</body>' +
           '</html>';
  }

  return '<!DOCTYPE html>' +
         '<html>' +
         '<head>' +
           '<meta charset="utf-8">' +
           '<meta name="viewport" content="width=device-width, initial-scale=1.0">' +
           '<style>' +
             '@media only screen and (max-width: 600px) {' +
               '.side-decor { display: none !important; width: 0 !important; }' +
               '.main-content-td { width: 100% !important; padding: 10px !important; }' +
               '.card-table { width: 100% !important; }' +
             '}' +
           '</style>' +
         '</head>' +
         '<body style="margin: 0; padding: 0; font-family: -apple-system, BlinkMacSystemFont, \'Segoe UI\', Roboto, Arial, sans-serif; -webkit-font-smoothing: antialiased; width: 100% !important;">' +
           '<table border="0" cellpadding="0" cellspacing="0" width="100%" style="padding: 30px 0;">' +
             '<tr>' +
               '<!-- Left Decor Column -->' +
               '<td class="side-decor" width="15%" align="right" valign="top" style="padding-top: 100px;">' +
                 '<div style="margin-bottom: 80px; position: relative;">' +
                   '<img src="https://img.icons8.com/ios/100/1d4ed8/speech-bubble.png" width="60" height="60" style="display: block; border: 0; opacity: 0.18; transform: rotate(-15deg); margin-right: -10px;" />' +
                 '</div>' +
                 '<div style="margin-bottom: 80px; position: relative;">' +
                   '<img src="https://img.icons8.com/ios/100/1d4ed8/speech-bubble.png" width="45" height="45" style="display: block; border: 0; opacity: 0.15; transform: rotate(10deg); margin-right: -5px;" />' +
                 '</div>' +
                 '<div style="margin-bottom: 80px; position: relative;">' +
                   '<img src="https://img.icons8.com/ios/100/1d4ed8/speech-bubble.png" width="55" height="55" style="display: block; border: 0; opacity: 0.16; transform: rotate(-5deg); margin-right: -8px;" />' +
                 '</div>' +
               '</td>' +
               
               '<!-- Center Main Card Column -->' +
               '<td class="main-content-td" width="600" align="center" valign="top">' +
                 
                 headerHtml +
                 
                 '<!-- Main Content Card -->' +
                 '<table class="card-table" border="0" cellpadding="0" cellspacing="0" width="600" style="max-width: 600px; background-color: #ffffff; border-radius: 16px; border: 1px solid #e2e8f0; box-shadow: 0 4px 12px rgba(15, 23, 42, 0.03); margin-bottom: 20px; text-align: left;">' +
                   '<tr>' +
                     '<td style="padding: 40px 30px; color: #334155; font-size: 16px; line-height: 1.6;">' +
                       '<p style="margin: 0 0 20px; font-size: 17px; font-weight: 600; color: #0f172a;">Hi ' + (recipientName || "there") + ',</p>' +
                       '<div style="font-size: 15px; color: #334155;">' + messageBody + '</div>' +
                     '</td>' +
                   '</tr>' +
                 '</table>' +
                 
                 footerHtml +
                 
               '</td>' +
               
               '<!-- Right Decor Column -->' +
               '<td class="side-decor" width="15%" align="left" valign="top" style="padding-top: 140px;">' +
                 '<div style="margin-bottom: 80px; position: relative;">' +
                   '<img src="https://img.icons8.com/ios/100/1d4ed8/speech-bubble.png" width="50" height="50" style="display: block; border: 0; opacity: 0.14; transform: rotate(20deg); margin-left: -8px;" />' +
                 '</div>' +
                 '<div style="margin-bottom: 80px; position: relative;">' +
                   '<img src="https://img.icons8.com/ios/100/1d4ed8/speech-bubble.png" width="55" height="55" style="display: block; border: 0; opacity: 0.16; transform: rotate(-10deg); margin-left: -10px;" />' +
                 '</div>' +
                 '<div style="margin-bottom: 80px; position: relative;">' +
                   '<img src="https://img.icons8.com/ios/100/1d4ed8/speech-bubble.png" width="40" height="40" style="display: block; border: 0; opacity: 0.12; transform: rotate(15deg); margin-left: -6px;" />' +
                 '</div>' +
               '</td>' +
             '</tr>' +
           '</table>' +
         '</body>' +
         '</html>';
}
