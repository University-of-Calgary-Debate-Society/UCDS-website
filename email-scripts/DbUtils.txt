/**
 * Ensures a sheet exists with the required columns. Creates it if missing.
 */
function getOrCreateSheet(spreadsheet, name, headers) {
  var sheet = spreadsheet.getSheetByName(name);
  if (!sheet) {
    sheet = spreadsheet.insertSheet(name);
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold");
    sheet.setFrozenRows(1);
  }
  return sheet;
}

/**
 * Finds and updates a row by its ID in a specific sheet.
 */
function updateRowById(sheet, idColIndex, idValue, newValuesObject, headers) {
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (data[i][idColIndex].toString() === idValue.toString()) {
      var rowIndex = i + 1;
      var rowRange = sheet.getRange(rowIndex, 1, 1, headers.length);
      var currentValues = data[i];
      var updatedValues = headers.map(function(header, idx) {
        if (newValuesObject.hasOwnProperty(header)) {
          return newValuesObject[header];
        }
        return currentValues[idx];
      });
      rowRange.setValues([updatedValues]);
      return true;
    }
  }
  return false;
}

/**
 * Finds and deletes a row by its ID.
 */
function deleteRowById(sheet, idColIndex, idValue) {
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (data[i][idColIndex].toString() === idValue.toString()) {
      sheet.deleteRow(i + 1);
      return true;
    }
  }
  return false;
}

/**
 * Returns a clean JSON string with appropriate headers.
 */
function jsonResponse(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
                       .setMimeType(ContentService.MimeType.JSON);
}
