/**
 * Helper function to select which spreadsheet to use
 */
function getMailingListSpreadsheet() {
  if (MAILING_LIST_SPREADSHEET_ID && MAILING_LIST_SPREADSHEET_ID !== "YOUR_MAILING_LIST_SPREADSHEET_ID_HERE") {
    try {
      return SpreadsheetApp.openById(MAILING_LIST_SPREADSHEET_ID);
    } catch(e) {
      console.error("Could not open mailing list spreadsheet by ID: " + e.toString() + ". Falling back to active spreadsheet.");
    }
  }
  return SpreadsheetApp.getActiveSpreadsheet();
}

/**
 * Save new subscribers in Google Sheets
 */
function handleSubscribe(p) {
  var email = p.email || "";
  var name = p.fullName || p.name || "";
  if (!email) {
    return ContentService.createTextOutput(JSON.stringify({ result: "error", error: "Email is required" }))
                         .setMimeType(ContentService.MimeType.JSON);
  }
  
  var spreadsheet = getMailingListSpreadsheet();
  var sheet = spreadsheet.getSheetByName("Subscribers");
  if (!sheet) {
    sheet = spreadsheet.insertSheet("Subscribers");
    sheet.appendRow(["Timestamp", "Email", "Name", "Status"]);
    sheet.getRange(1, 1, 1, 4).setFontWeight("bold");
    sheet.setFrozenRows(1);
  }
  
  var data = sheet.getDataRange().getValues();
  var rowIndex = -1;
  for (var i = 1; i < data.length; i++) {
    if (data[i][1].toString().trim().toLowerCase() === email.trim().toLowerCase()) {
      rowIndex = i + 1;
      break;
    }
  }
  
  if (rowIndex !== -1) {
    sheet.getRange(rowIndex, 3).setValue(name); 
    sheet.getRange(rowIndex, 4).setValue("Active"); 
  } else {
    sheet.appendRow([new Date(), email, name, "Active"]);
  }
  
  return ContentService.createTextOutput(JSON.stringify({ result: "success" }))
                       .setMimeType(ContentService.MimeType.JSON);
}

/**
 * Set up custom Sheet menus to trigger news dispatching
 */
function onOpen() {
  var ui = SpreadsheetApp.getUi();
  ui.createMenu('UCDS Tools')
      .addItem('Send Newsletter', 'promptNewsletter')
      .addToUi();
}

function promptNewsletter() {
  var ui = SpreadsheetApp.getUi();
  
  var subjectPrompt = ui.prompt('Send Newsletter', 'Enter Email Subject:', ui.ButtonSet.OK_CANCEL);
  if (subjectPrompt.getSelectedButton() !== ui.Button.OK) return;
  var subject = subjectPrompt.getResponseText().trim();
  if (!subject) {
    ui.alert("Subject cannot be empty!");
    return;
  }

  var bodyPrompt = ui.prompt('Send Newsletter', 'Enter HTML Body/Message:', ui.ButtonSet.OK_CANCEL);
  if (bodyPrompt.getSelectedButton() !== ui.Button.OK) return;
  var bodyContent = bodyPrompt.getResponseText().trim();
  if (!bodyContent) {
    ui.alert("Email content cannot be empty!");
    return;
  }

  var confirm = ui.alert('Confirm Send', 'Are you sure you want to send this newsletter to all active subscribers?', ui.ButtonSet.YES_NO);
  if (confirm === ui.Button.YES) {
    sendNewsletter(subject, bodyContent);
    ui.alert("Newsletter dispatched successfully!");
  }
}

/**
 * Newsletter Dispatcher Function
 */
function sendNewsletter(subject, messageBody) {
  var sheet = getMailingListSpreadsheet().getSheetByName("Subscribers");
  if (!sheet) return;
  
  var data = sheet.getDataRange().getValues();
  
  var webAppUrl = WEB_APP_URL;
  
  for (var i = 1; i < data.length; i++) {
    var email = data[i][1];
    var name = data[i][2];
    var status = data[i][3];
    
    if (status === "Active") {
      var unsubscribeLink = webAppUrl + "?action=unsubscribe&email=" + encodeURIComponent(email);
      var personalizedBody = "Hi " + (name || "there") + ",\n\n" + messageBody + 
                             "\n\n---\nTo unsubscribe, click here: " + unsubscribeLink;
      
      var htmlBody = '<div style="font-family: -apple-system, sans-serif; max-width: 560px; margin: 0 auto; color: #1e293b; line-height: 1.6;">' +
                       '<div style="border-bottom: 2px solid #112854; padding-bottom: 12px; margin-bottom: 20px;">' +
                         '<h3 style="color: #112854; margin: 0; font-size: 1rem; text-transform: uppercase;">University of Calgary Debate Society</h3>' +
                       '</div>' +
                       '<p>Hi ' + (name || "there") + ',</p>' +
                       '<div>' + messageBody + '</div>' +
                       '<div style="margin-top: 40px; border-top: 1px solid #e2e8f0; padding-top: 16px; font-size: 0.8rem; color: #64748b;">' +
                         'You received this email because you are subscribed to the UCDS newsletter list. ' +
                         '<a href="' + unsubscribeLink + '" style="color: #2563eb; text-decoration: underline;">Click here to unsubscribe</a>.' +
                       '</div>' +
                     '</div>';
      
      try {
        MailApp.sendEmail({
          to: email,
          subject: subject,
          body: personalizedBody,
          htmlBody: htmlBody
        });
      } catch (err) {
        console.error("Failed to mail " + email + ": " + err.toString());
      }
    }
  }
}
